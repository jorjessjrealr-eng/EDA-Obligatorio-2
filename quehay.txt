typedef struct nodo {
  Paquete dato;
  struct nodo *izquierda;
  struct nodo *derecha;
} Nodo;

struct abb {
  Nodo *raiz;
};

Abb *plantarABB(void) {
  Abb *arbol = malloc(sizeof(Abb));
  if (arbol != NULL) {
    arbol->raiz = NULL;
  }
  return arbol;
}

static void podarNodos(Nodo *nodo) {
  if (nodo != NULL) {
    podarNodos(nodo->izquierda);
    podarNodos(nodo->derecha);
    free(nodo);
  }
}

void talarABB(Abb *arbol) {
  if (arbol != NULL) {
    podarNodos(arbol->raiz);
    free(arbol);
  }
}

static Nodo *crearNodo(const Paquete *paquete) {
  Nodo *nodo = malloc(sizeof(Nodo));
  if (nodo != NULL) {
    nodo->dato = *paquete;
    nodo->izquierda = NULL;
    nodo->derecha = NULL;
  }
  return nodo;
}

static Nodo *injertarNodo(Nodo *nodo, const Paquete *paquete,
                          bool *yaInsertado) {
  if (nodo == NULL) {
    Nodo *nuevo = crearNodo(paquete);
    *yaInsertado = (nuevo != NULL);
    return nuevo;
  }
  int comparacion = strcmp(paquete->trackingID, nodo->dato.trackingID);
  if (comparacion < 0) {
    nodo->izquierda = injertarNodo(nodo->izquierda, paquete, yaInsertado);
  } else if (comparacion > 0) {
    nodo->derecha = injertarNodo(nodo->derecha, paquete, yaInsertado);
  } else {
    *yaInsertado = false;
  }
  return nodo;
}

bool injertarPaquete(Abb *arbol, const Paquete *paquete) {
  if (arbol == NULL || paquete == NULL)
    return false;
  bool yaInsertado = false;
  arbol->raiz = injertarNodo(arbol->raiz, paquete, &yaInsertado);
  return yaInsertado;
}

static const Nodo *buscarNodo(const Nodo *nodo, const char *trackingID) {
  if (nodo == NULL)
    return NULL;
  int comparacion = strcmp(trackingID, nodo->dato.trackingID);
  if (comparacion == 0)
    return nodo;
  if (comparacion < 0)
    return buscarNodo(nodo->izquierda, trackingID);
  return buscarNodo(nodo->derecha, trackingID);
}

bool buscarPorID(const Abb *arbol, const char *trackingID, Paquete *salida) {
  if (arbol == NULL || trackingID == NULL || salida == NULL)
    return false;
  const Nodo *nodo = buscarNodo(arbol->raiz, trackingID);
  if (nodo == NULL)
    return false;
  *salida = nodo->dato;
  return true;
}

static void enOrden(const Nodo *nodo) {
  if (nodo != NULL) {
    enOrden(nodo->izquierda);
    printf("%s | %s | %.2f kg | $%.2f\n", nodo->dato.trackingID,
           nodo->dato.destino, nodo->dato.peso, nodo->dato.costoEnvio);
    enOrden(nodo->derecha);
  }
}

void recorrerEnOrden(const Abb *arbol) {
  if (arbol != NULL) {
    enOrden(arbol->raiz);
  }
}

static int profundidad(const Nodo *nodo) {
  if (nodo == NULL)
    return 0;
  int profundidadIzquierda = profundidad(nodo->izquierda);
  int profundidadDerecha = profundidad(nodo->derecha);
  if (profundidadIzquierda > profundidadDerecha) {
    return 1 + profundidadIzquierda;
  }
  return 1 + profundidadDerecha;
}

int profundidadABB(const Abb *arbol) {
  if (arbol == NULL)
    return 0;
  return profundidad(arbol->raiz);
}

static float pesoTotal(const Nodo *nodo) {
  if (nodo == NULL)
    return 0;
  return nodo->dato.peso + pesoTotal(nodo->izquierda) +
         pesoTotal(nodo->derecha);
}

float pesoTotalABB(const Abb *arbol) {
  if (arbol == NULL)
    return 0;
  return pesoTotal(arbol->raiz);
}

static const Nodo *buscarCosto(const Nodo *nodo, float costoMin) {
  if (nodo == NULL)
    return NULL;
  const Nodo *resultadoIzquierda = buscarCosto(nodo->izquierda, costoMin);
  if (resultadoIzquierda != NULL)
    return resultadoIzquierda;
  if (nodo->dato.costoEnvio >= costoMin)
    return nodo;
  return buscarCosto(nodo->derecha, costoMin);
}

bool buscarPorCosto(const Abb *arbol, float costoMin, Paquete *salida) {
  if (arbol == NULL || salida == NULL)
    return false;
  const Nodo *nodo = buscarCosto(arbol->raiz, costoMin);
  if (nodo == NULL)
    return false;
  *salida = nodo->dato;
  return true;
}

struct monton {
  Envio *datos;
  int tamano;
  int capacidad;
};

Monton *crearMonton(int capacidad) {
  if (capacidad <= 0) {
    capacidad = 10;
  }
  Monton *monton = malloc(sizeof(Monton));
  if (monton == NULL)
    return NULL;
  monton->datos = malloc(capacidad * sizeof(Envio));
  if (monton->datos == NULL) {
    free(monton);
    return NULL;
  }
  monton->tamano = 0;
  monton->capacidad = capacidad;
  return monton;
}

void destruirMonton(Monton *monton) {
  if (monton != NULL) {
    free(monton->datos);
    free(monton);
  }
}

bool montonVacio(const Monton *monton) {
  return monton == NULL || monton->tamano == 0;
}

static void intercambiar(Envio *primero, Envio *segundo) {
  Envio auxiliar = *primero;
  *primero = *segundo;
  *segundo = auxiliar;
}

static void subir(Monton *monton, int indice) {
  while (indice > 0 && monton->datos[indice].nivelUrgencia >
                           monton->datos[(indice - 1) / 2].nivelUrgencia) {
    int padre = (indice - 1) / 2;
    intercambiar(&monton->datos[indice], &monton->datos[padre]);
    indice = padre;
  }
}

static void bajar(Monton *monton, int indice) {
  bool seguir = true;
  while (seguir) {
    int izquierda = 2 * indice + 1;
    int derecha = 2 * indice + 2;
    int mayor = indice;
    if (izquierda < monton->tamano && monton->datos[izquierda].nivelUrgencia >
                                          monton->datos[mayor].nivelUrgencia) {
      mayor = izquierda;
    }
    if (derecha < monton->tamano && monton->datos[derecha].nivelUrgencia >
                                        monton->datos[mayor].nivelUrgencia) {
      mayor = derecha;
    }
    if (mayor != indice) {
      intercambiar(&monton->datos[indice], &monton->datos[mayor]);
      indice = mayor;
    } else {
      seguir = false;
    }
  }
}

bool insertarEnvio(Monton *monton, const Envio *envio) {
  if (monton == NULL || envio == NULL)
    return false;
  if (monton->tamano == monton->capacidad) {
    Envio *nuevo =
        realloc(monton->datos, monton->capacidad * 2 * sizeof(Envio));
    if (nuevo == NULL)
      return false;
    monton->datos = nuevo;
    monton->capacidad *= 2;
  }
  monton->datos[monton->tamano] = *envio;
  subir(monton, monton->tamano);
  monton->tamano++;
  return true;
}

bool consultarMaxUrgencia(const Monton *monton, Envio *salida) {
  if (montonVacio(monton) || salida == NULL)
    return false;
  *salida = monton->datos[0];
  return true;
}

bool extraerMaximo(Monton *monton, Envio *salida) {
  if (montonVacio(monton))
    return false;
  if (salida != NULL) {
    *salida = monton->datos[0];
  }
  monton->tamano--;
  monton->datos[0] = monton->datos[monton->tamano];
  bajar(monton, 0);
  return true;
}

typedef struct nodo {
  Operacion dato;
  struct nodo *siguiente;
} Nodo;

struct pila {
  Nodo *cima;
};

Pila *crearPila(void) {
  Pila *pila = malloc(sizeof(Pila));
  if (pila != NULL) {
    pila->cima = NULL;
  }
  return pila;
}

bool pilaVacia(const Pila *pila) { return pila == NULL || pila->cima == NULL; }

bool apilar(Pila *pila, const Operacion *operacion) {
  if (pila == NULL || operacion == NULL)
    return false;
  Nodo *nuevo = malloc(sizeof(Nodo));
  if (nuevo == NULL)
    return false;
  nuevo->dato = *operacion;
  nuevo->siguiente = pila->cima;
  pila->cima = nuevo;
  return true;
}

bool desapilar(Pila *pila, Operacion *salida) {
  if (pilaVacia(pila))
    return false;
  Nodo *borrar = pila->cima;
  if (salida != NULL) {
    *salida = borrar->dato;
  }
  pila->cima = borrar->siguiente;
  free(borrar);
  return true;
}

bool consultarCima(const Pila *pila, Operacion *cima) {
  if (pilaVacia(pila) || cima == NULL)
    return false;
  *cima = pila->cima->dato;
  return true;
}

void destruirPila(Pila *pila) {
  if (pila != NULL) {
    while (!pilaVacia(pila)) {
      desapilar(pila, NULL);
    }
    free(pila);
  }
}
